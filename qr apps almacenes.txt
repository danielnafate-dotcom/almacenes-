    ''' <summary>
    ''' Define el esquema de la tabla de datos principal en memoria.
    ''' </summary>
    Private Function CrearEstructuraTabla() As DataTable
        Dim dt As New DataTable("PackingList")
        dt.Columns.Add("Fecha Carga", GetType(String))
        dt.Columns.Add("Numero Consecutivo", GetType(String))
        dt.Columns.Add("Packing List No.", GetType(String))
        dt.Columns.Add("Sr. No.", GetType(String))
        dt.Columns.Add("Part No.", GetType(String))
        dt.Columns.Add("Descrtion", GetType(String))
        dt.Columns.Add("Qty", GetType(String))
        dt.Columns.Add("Case No.", GetType(String))
        dt.Columns.Add("Ubicacion", GetType(String))
        Return dt
    End Function

    ''' <summary>
    ''' Carga el historial de datos almacenado localmente en XML.
    ''' </summary>
    Private Sub CargarHistoricoXML()
        If File.Exists(RUTA_XML) Then
            Try
                Dim dt As New DataTable()
                dt.ReadXml(RUTA_XML)
                If Not dt.Columns.Contains("Ubicacion") Then
                    dt.Columns.Add("Ubicacion", GetType(String))
                End If
                dgvDatos.DataSource = dt
                lblInfo.Text = "Histórico cargado: " & dt.Rows.Count & " registros."
            Catch ex As Exception
                dgvDatos.DataSource = CrearEstructuraTabla()
            End Try
        Else
            dgvDatos.DataSource = CrearEstructuraTabla()
        End If
    End Sub

    ''' <summary>
    ''' Guarda la información del DataGridView en el archivo XML local.
    ''' </summary>
    Private Sub GuardarHistorico()
        If dgvDatos.DataSource IsNot Nothing Then
            Dim dt As DataTable = CType(dgvDatos.DataSource, DataTable)
            dt.WriteXml(RUTA_XML, XmlWriteMode.WriteSchema)
        End If
    End Sub

    ''' <summary>
    ''' Realiza la lectura vía OLEDB, parseo de filas y extracción de Packing List.
    ''' </summary>
    Private Sub ProcesarArchivoExcel(rutaExcel As String, fechaCarga As String, consecutivoInicial As Integer)
        Dim rutaTemporal As String = Path.Combine(Path.GetTempPath(), "temp_" & Guid.NewGuid().ToString() & Path.GetExtension(rutaExcel))

        Try
            File.Copy(rutaExcel, rutaTemporal, True)

            Dim dtExistente As DataTable
            If dgvDatos.DataSource IsNot Nothing Then
                dtExistente = CType(dgvDatos.DataSource, DataTable)
            Else
                dtExistente = CrearEstructuraTabla()
            End If

            If Not dtExistente.Columns.Contains("Ubicacion") Then
                dtExistente.Columns.Add("Ubicacion", GetType(String))
            End If

            Dim cs As String = ""
            Dim ext As String = Path.GetExtension(rutaExcel).ToLower()

            If ext = ".xlsx" Then
                cs = "Provider=Microsoft.ACE.OLEDB.12.0;Data Source=" & rutaTemporal & ";Extended Properties='Excel 12.0 Xml;HDR=NO;IMEX=1;Mode=Share Deny None;'"
            ElseIf ext = ".xls" Then
                cs = "Provider=Microsoft.ACE.OLEDB.12.0;Data Source=" & rutaTemporal & ";Extended Properties='Excel 8.0;HDR=NO;IMEX=1;Mode=Share Deny None;'"
            Else
                MessageBox.Show("Formato no soportado.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
                Exit Sub
            End If

            Using conn As New System.Data.OleDb.OleDbConnection(cs)
                conn.Open()
                Dim dtSchema As DataTable = conn.GetOleDbSchemaTable(System.Data.OleDb.OleDbSchemaGuid.Tables, Nothing)

                If dtSchema IsNot Nothing AndAlso dtSchema.Rows.Count > 0 Then
                    Dim nombreHoja As String = ""

                    For Each rowSchema As DataRow In dtSchema.Rows
                        Dim tempName As String = rowSchema("TABLE_NAME").ToString()
                        If tempName.EndsWith("$") OrElse tempName.EndsWith("$'") Then
                            nombreHoja = tempName.Trim("'"c)
                            Exit For
                        End If
                    Next

                    If String.IsNullOrEmpty(nombreHoja) Then
                        nombreHoja = dtSchema.Rows(0)("TABLE_NAME").ToString().Trim("'"c)
                    End If

                    Dim query As String = "SELECT * FROM [" & nombreHoja & "]"

                    Using cmd As New System.Data.OleDb.OleDbCommand(query, conn)
                        Using da As New System.Data.OleDb.OleDbDataAdapter(cmd)
                            Dim dtExcel As New DataTable()
                            da.Fill(dtExcel)

                            Dim registrosAgregados As Integer = 0
                            Dim currentPackingList As String = ""
                            Dim currentCaseNo As String = ""
                            Dim numeroConsecutivoActual As Integer = consecutivoInicial

                            Dim mapaConsecutivosCaja As New Dictionary(Of String, Integer)()

                            For Each row As DataRow In dtExcel.Rows
                                Dim textoFilaCompleta As String = ""
                                For colIdx As Integer = 0 To row.Table.Columns.Count - 1
                                    If Not IsDBNull(row(colIdx)) Then
                                        textoFilaCompleta &= row(colIdx).ToString() & " "
                                    End If
                                Next

                                Dim textoUpper As String = textoFilaCompleta.ToUpper()

                                ' Identificar encabezado de Packing List No.
                                If textoUpper.Contains("PACKING LIST NO") Then
                                    Dim match As Match = Regex.Match(textoFilaCompleta, "(?i)Packing\s*list\s*No\.?\s*(\d+)")
                                    If match.Success Then
                                        currentPackingList = match.Groups(1).Value
                                    Else
                                        Dim matchNum As Match = Regex.Match(textoFilaCompleta, "\d{6,}")
                                        If matchNum.Success Then
                                            currentPackingList = matchNum.Value
                                        End If
                                    End If
                                    Continue For
                                End If

                                ' Identificar número de caja o Bulto (Case Box Number)
                                If textoUpper.Contains("CASE BOX NUMBER") Then
                                    Dim matchCase As Match = Regex.Match(textoFilaCompleta, "(?i)Case\s*Box\s*Number\s*(\d+)\s*/\s*(\d+)")
                                    If matchCase.Success Then
                                        currentCaseNo = "Case " & matchCase.Groups(1).Value & " of " & matchCase.Groups(2).Value
                                    Else
                                        currentCaseNo = textoFilaCompleta.Trim()
                                    End If
                                    Continue For
                                End If

                                Dim col0 As String = If(row.Table.Columns.Count > 0 AndAlso Not IsDBNull(row(0)), row(0).ToString().Trim(), "")
                                Dim tempSr As Integer

                                ' Si la fila empieza por un número consecutivo de ítem (Sr. No.)
                                If Integer.TryParse(col0, tempSr) Then
                                    Dim partNo As String = If(row.Table.Columns.Count > 1 AndAlso Not IsDBNull(row(1)), row(1).ToString().Trim(), "")

                                    Dim idxQty As Integer = -1
                                    Dim qtyVal As String = ""

                                    For colIdx As Integer = row.Table.Columns.Count - 1 To 2 Step -1
                                        If Not IsDBNull(row(colIdx)) AndAlso IsNumeric(row(colIdx).ToString().Trim()) Then
                                            qtyVal = row(colIdx).ToString().Trim()
                                            idxQty = colIdx
                                            Exit For
                                        End If
                                    Next

                                    Dim sbDesc As New StringBuilder()
                                    Dim finDesc As Integer = If(idxQty > 2, idxQty - 1, row.Table.Columns.Count - 1)

                                    For colIdx As Integer = 2 To finDesc
                                        If Not IsDBNull(row(colIdx)) Then
                                            Dim valCell As String = row(colIdx).ToString().Trim()
                                            If Not String.IsNullOrEmpty(valCell) AndAlso Not IsNumeric(valCell) Then
                                                sbDesc.Append(valCell).Append(" ")
                                            End If
                                        End If
                                    Next

                                    ' Mapeo de consecutivo único por caja
                                    Dim consecutivoAAsignar As Integer
                                    If mapaConsecutivosCaja.ContainsKey(currentCaseNo) Then
                                        consecutivoAAsignar = mapaConsecutivosCaja(currentCaseNo)
                                    Else
                                        consecutivoAAsignar = numeroConsecutivoActual
                                        mapaConsecutivosCaja.Add(currentCaseNo, consecutivoAAsignar)
                                        numeroConsecutivoActual += 1
                                    End If

                                    Dim nRow As DataRow = dtExistente.NewRow()
                                    nRow("Fecha Carga") = fechaCarga
                                    nRow("Numero Consecutivo") = consecutivoAAsignar.ToString()
                                    nRow("Packing List No.") = currentPackingList
                                    nRow("Sr. No.") = col0
                                    nRow("Part No.") = partNo
                                    nRow("Descrtion") = sbDesc.ToString().Trim()
                                    nRow("Qty") = qtyVal
                                    nRow("Case No.") = currentCaseNo
                                    nRow("Ubicacion") = ""
                                    dtExistente.Rows.Add(nRow)

                                    registrosAgregados += 1
                                End If
                            Next

                            dgvDatos.DataSource = dtExistente
                            GuardarHistorico()
                            lblInfo.Text = "Histórico: " & dtExistente.Rows.Count & " registros."
                            MessageBox.Show("Se importaron " & registrosAgregados & " registros válidos. Las cajas repetidas conservan su mismo consecutivo.", "Importación Exitosa", MessageBoxButtons.OK, MessageBoxIcon.Information)
                        End Using
                    End Using
                End If
            End Using

        Catch ex As Exception
            MessageBox.Show("Error al leer Excel: " & ex.Message, "Error de Lectura", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Finally
            If File.Exists(rutaTemporal) Then
                Try
                    File.Delete(rutaTemporal)
                Catch ex As Exception
                    ' Ignorar error de borrado de temporal
                End Try
            End If
        End Try
    End Sub

    ' =========================================================================
    ''' <summary>
    ''' Consulta la API Web App de Google Apps Script vía HTTP GET.
    ''' </summary>
    Private Sub btnCargarGoogleSheets_Click(sender As Object, e As EventArgs) Handles btnCargarGoogleSheets.Click
        btnCargarGoogleSheets.Enabled = False
        btnCargarGoogleSheets.Text = "CARGANDO..."

        Try
            ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12
            Dim request As HttpWebRequest = CType(WebRequest.Create(URL_WEB_APP), HttpWebRequest)
            request.Method = "GET"
            request.UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"

            Using response As HttpWebResponse = CType(request.GetResponse(), HttpWebResponse)
                Using reader As New StreamReader(response.GetResponseStream(), Encoding.UTF8)
                    Dim jsonResponse As String = reader.ReadToEnd()

                    If response.StatusCode = HttpStatusCode.OK Then
                        ProcesarJSONGoogleSheets(jsonResponse)
                    Else
                        MessageBox.Show("Error al obtener datos: " & response.StatusCode.ToString(), "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
                    End If
                End Using
            End Using

        Catch ex As Exception
            MessageBox.Show("Error al conectar con Google Sheets: " & ex.Message, "Error de Conexión", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Finally
            btnCargarGoogleSheets.Enabled = True
            btnCargarGoogleSheets.Text = "Cargar GSheets"
        End Try
    End Sub

    ''' <summary>
    ''' Parsea la respuesta JSON enviada por Google Apps Script.
    ''' </summary>
    Private Sub ProcesarJSONGoogleSheets(json As String)
        Try
            Dim dtExistente As DataTable
            If dgvDatos.DataSource IsNot Nothing Then
                dtExistente = CType(dgvDatos.DataSource, DataTable)
            Else
                dtExistente = CrearEstructuraTabla()
            End If

            If Not dtExistente.Columns.Contains("Ubicacion") Then
                dtExistente.Columns.Add("Ubicacion", GetType(String))
            End If

            Dim jsonLimpio As String = json.Trim()
            If jsonLimpio.StartsWith("[") AndAlso jsonLimpio.EndsWith("]") Then
                jsonLimpio = jsonLimpio.Substring(1, jsonLimpio.Length - 2).Trim()
            End If

            If String.IsNullOrWhiteSpace(jsonLimpio) Then
                MessageBox.Show("No se encontraron registros en la hoja de Google Sheets.", "Sin Datos", MessageBoxButtons.OK, MessageBoxIcon.Information)
                Exit Sub
            End If

            Dim elementos() As String = Regex.Split(jsonLimpio, "\}\s*,\s*\{")
            Dim cargados As Integer = 0

            For Each elem As String In elementos
                Dim objetoStr As String = elem.Replace("{", "").Replace("}", "")

                Dim fecha As String = ExtraerValorJSON(objetoStr, "fecha")
                Dim consecutivo As String = ExtraerValorJSON(objetoStr, "consecutivo")
                Dim packingList As String = ExtraerValorJSON(objetoStr, "packingList")
                Dim srNo As String = ExtraerValorJSON(objetoStr, "srNo")
                Dim partNo As String = ExtraerValorJSON(objetoStr, "partNo")
                Dim desc As String = ExtraerValorJSON(objetoStr, "desc")
                Dim qty As String = ExtraerValorJSON(objetoStr, "qty")
                Dim caseNo As String = ExtraerValorJSON(objetoStr, "caseNo")
                Dim ubicacion As String = ExtraerValorJSON(objetoStr, "ubicacion")

                Dim nRow As DataRow = dtExistente.NewRow()
                nRow("Fecha Carga") = fecha
                nRow("Numero Consecutivo") = consecutivo
                nRow("Packing List No.") = packingList
                nRow("Sr. No.") = srNo
                nRow("Part No.") = partNo
                nRow("Descrtion") = desc
                nRow("Qty") = qty
                nRow("Case No.") = caseNo
                nRow("Ubicacion") = ubicacion

                dtExistente.Rows.Add(nRow)
                cargados += 1
            Next

            dgvDatos.DataSource = dtExistente
            GuardarHistorico()
            lblInfo.Text = "Histórico: " & dtExistente.Rows.Count & " registros."
            MessageBox.Show("Se cargaron exitosamente " & cargados & " registros desde Google Sheets.", "Carga Exitosa", MessageBoxButtons.OK, MessageBoxIcon.Information)

        Catch ex As Exception
            MessageBox.Show("Error al interpretar los datos de Google Sheets: " & ex.Message, "Error de Lectura", MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try
    End Sub

    ''' <summary>
    ''' Auxiliar para extraer propiedades de un string de objeto JSON.
    ''' </summary>
    Private Function ExtraerValorJSON(jsonChunk As String, clave As String) As String
        Dim patron As String = """" & clave & """\s*:\s*""(.*?)"""
        Dim match As Match = Regex.Match(jsonChunk, patron)
        If match.Success Then
            Return match.Groups(1).Value.Replace("\""", """")
        End If
        Return ""
    End Function

    ''' <summary>
    ''' Genera y descarga una imagen QR utilizando la API qrserver.com.
    ''' </summary>
    Private Function ObtenerImagenQR(url As String) As Image
        Try
            Dim apiUrl As String = "https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=" & Uri.EscapeDataString(url)
            Using client As New WebClient()
                Dim bytes() As Byte = client.DownloadData(apiUrl)
                Using ms As New MemoryStream(bytes)
                    Return New Bitmap(Image.FromStream(ms))
                End Using
            End Using
        Catch ex As Exception
            Dim bmp As New Bitmap(200, 200)
            Using g As Graphics = Graphics.FromImage(bmp)
                g.Clear(Color.White)
                g.DrawRectangle(Pens.Red, 2, 2, 196, 196)
                g.DrawString("Sin QR", New Font("Arial", 10, FontStyle.Bold), Brushes.Red, New PointF(70, 90))
            End Using
            Return bmp
        End Try
    End Function

    ''' <summary>
    ''' Sincroniza la tabla local enviando un payload JSON hacia Google Apps Script.
    ''' </summary>
    Private Sub btnExportarGoogle_Click(sender As Object, e As EventArgs) Handles btnExportarGoogle.Click
        If dgvDatos.Rows.Count = 0 Then
            MessageBox.Show("No hay datos para exportar.", "Atención", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Return
        End If

        btnExportarGoogle.Enabled = False
        btnExportarGoogle.Text = "EXPORTANDO..."

        Try
            Dim dt As DataTable = CType(dgvDatos.DataSource, DataTable)
            Dim jsonRows As New List(Of String)()

            For Each row As DataRow In dt.Rows
                Dim fecha As String = row("Fecha Carga").ToString().Replace("""", "\""")
                Dim consecutivo As String = row("Numero Consecutivo").ToString().Replace("""", "\""")
                Dim packingList As String = row("Packing List No.").ToString().Replace("""", "\""")
                Dim srNo As String = row("Sr. No.").ToString().Replace("""", "\""")
                Dim partNo As String = row("Part No.").ToString().Replace("""", "\""")
                Dim desc As String = row("Descrtion").ToString().Replace("""", "\""")
                Dim qty As String = row("Qty").ToString().Replace("""", "\""")
                Dim caseNo As String = row("Case No.").ToString().Replace("""", "\""")
                Dim ubicacion As String = If(dt.Columns.Contains("Ubicacion"), row("Ubicacion").ToString().Replace("""", "\"""), "")

                Dim item As String = String.Format("{{\""fecha\"":\""{0}\"",\""consecutivo\"":\""{1}\"",\""packingList\"":\""{2}\"",\""srNo\"":\""{3}\"",\""partNo\"":\""{4}\"",\""desc\"":\""{5}\"",\""qty\"":\""{6}\"",\""caseNo\"":\""{7}\"",\""ubicacion\"":\""{8}\""}}",
                        fecha, consecutivo, packingList, srNo, partNo, desc, qty, caseNo, ubicacion)
                jsonRows.Add(item)
            Next

            Dim jsonPayload As String = "[" & String.Join(",", jsonRows) & "]"

            ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12

            Dim request As HttpWebRequest = CType(WebRequest.Create(URL_WEB_APP), HttpWebRequest)
            request.Method = "POST"
            request.ContentType = "application/json; charset=utf-8"
            request.AllowAutoRedirect = True
            request.UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"

            Dim byteArray As Byte() = Encoding.UTF8.GetBytes(jsonPayload)
            request.ContentLength = byteArray.Length

            Using dataStream As Stream = request.GetRequestStream()
                dataStream.Write(byteArray, 0, byteArray.Length)
            End Using

            Using response As HttpWebResponse = CType(request.GetResponse(), HttpWebResponse)
                Using reader As New StreamReader(response.GetResponseStream(), Encoding.UTF8)
                    Dim respuestaServidor As String = reader.ReadToEnd()

                    If response.StatusCode = HttpStatusCode.OK Then
                        MessageBox.Show("Sincronización con Google Sheets exitosa.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information)
                    Else
                        MessageBox.Show("Error en el servidor: " & response.StatusCode.ToString(), "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
                    End If
                End Using
            End Using

        Catch ex As WebException
            If ex.Response IsNot Nothing Then
                Using reader As New StreamReader(ex.Response.GetResponseStream())
                    Dim errorText As String = reader.ReadToEnd()
                    MessageBox.Show("Error de Autorización HTTP: " & errorText, "Error 401/HTTP", MessageBoxButtons.OK, MessageBoxIcon.Error)
                End Using
            Else
                MessageBox.Show("Error al exportar a Google Sheets: " & ex.Message, "Error de Red", MessageBoxButtons.OK, MessageBoxIcon.Error)
            End If
        Catch ex As Exception
            MessageBox.Show("Error inesperado: " & ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
        Finally
            btnExportarGoogle.Enabled = True
            btnExportarGoogle.Text = "2. Exportar Google"
        End Try
    End Sub

                    .LineAlignment = StringAlignment.Near,

            Dim posYData As Integer = tablaTop + headerHeight + 5

    ''' <summary>
    ''' Descarga y asigna la imagen QR generada al PictureBox de vista previa.
    ''' </summary>
    Private Sub CargarQRVistaPrevia(urlFicha As String)
        Try
            Dim apiUrl As String = "https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=" & Uri.EscapeDataString(urlFicha)
            Using client As New WebClient()
                Dim bytes() As Byte = client.DownloadData(apiUrl)
                Using ms As New MemoryStream(bytes)
                    Dim imgTemp As Image = Image.FromStream(ms)
                    If picQR.Image IsNot Nothing Then picQR.Image.Dispose()
                    picQR.Image = New Bitmap(imgTemp)
                End Using
            End Using
        Catch ex As Exception
            If picQR.Image IsNot Nothing Then picQR.Image.Dispose()
            picQR.Image = Nothing
        End Try
    End Sub

Public Module ModuloFormatos
    ''' <summary>
    ''' Función de soporte para formatear strings de texto limpio para exportaciones CSV/JSON.
    ''' </summary>
    Public Function FormatearTextoLimpio(input As String) As String
        If String.IsNullOrEmpty(input) Then Return ""
        Return input.Replace(Chr(34), "\" & Chr(34)).Replace(vbCrLf, " ").Replace(vbCr, " ").Replace(vbLf, " ")
    End Function

    ''' <summary>
    ''' Función de validación de entradas numéricas integradas.
    ''' </summary>
    Public Function EsNumeroValido(valor As String) As Boolean
        Dim n As Double
        Return Double.TryParse(valor, n)
    End Function
End Module

Public Class LogAuditoria
    Private Shared ReadOnly RUTA_LOG As String = Path.Combine(Application.StartupPath, "bitacora_sistema.log")

    ''' <summary>
    ''' Esconde o guarda operaciones de registro dentro de la bitácora del ejecutable.
    ''' </summary>
    Public Shared Sub RegistrarEvento(mensaje As String)
        Try
            Dim linea As String = String.Format("[{0}] {1}", DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss"), mensaje)
            File.AppendAllText(RUTA_LOG, linea & Environment.NewLine)
        Catch ex As Exception
            ' Evitar interrupción de hilo principal en fallos de E/S de logs
        End Try
    End Sub
End Class

' CLASE AUXILIAR PARA LA MANIPULACIÓN Y FORMATEO DE DATO DE ENTRADA
' =============================================================================

Public Class ManejadorEstructuras
    ''' <summary>
    ''' Verifica la compatibilidad de esquemas entre DataTables dinámicas.
    ''' </summary>
    Public Shared Sub ValidarEsquemaTabla(ByRef dt As DataTable)
        If dt IsNot Nothing Then
            Dim columnasRequeridas() As String = {"Fecha Carga", "Numero Consecutivo", "Packing List No.", "Sr. No.", "Part No.", "Descrtion", "Qty", "Case No.", "Ubicacion"}
            For Each colName As String In columnasRequeridas
                If Not dt.Columns.Contains(colName) Then
                    dt.Columns.Add(colName, GetType(String))
                End If
            Next
        End If
    End Sub
End Class
